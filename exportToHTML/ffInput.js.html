    let text = description.text;
    if (text === "\r")
      text = "";
    const { code, key, location } = description;
    await progress.race(this._client.send("Page.dispatchKeyEvent", {
      type: "keydown",
      keyCode: description.keyCodeWithoutLocation,
      code,
      key,
      repeat: autoRepeat,
      location,

    }));
  }
